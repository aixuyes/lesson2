cmake_minimum_required(VERSION 3.16)
project(lesson2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(zenohc 1.10.1 EXACT REQUIRED)
find_package(zenohcxx 1.10.1 EXACT REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs highgui)

add_executable(pub pub.cpp)
target_compile_features(pub PRIVATE cxx_std_17)
target_include_directories(pub PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(pub PRIVATE ${OpenCV_LIBS})
target_link_libraries(pub PRIVATE zenohcxx::zenohc Threads::Threads)
add_executable(sub sub.cpp)
target_compile_features(sub PRIVATE cxx_std_17)
target_include_directories(sub PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(sub PRIVATE ${OpenCV_LIBS})
target_link_libraries(sub PRIVATE zenohcxx::zenohc Threads::Threads)
